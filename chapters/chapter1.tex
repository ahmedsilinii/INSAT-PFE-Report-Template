\subsection{Insérer un tableau}

% Table style: caption ABOVE the tabular, booktabs rules, bold header row,
% @{} at both ends so the rules line up with the text, \small text.
Le titre d'un tableau se place au-dessus, celui d'une figure en dessous. Le
\autoref{tab:ch1-text} montre un tableau de texte.

% Text table: p{} columns wrap long cells; their widths add up to the text
% width minus the column gaps.
\begin{table}[H]
  \centering
  \caption{Exemple de tableau de texte : remplacez-le par le vôtre.}
  \label{tab:ch1-text}

  \begin{tabular}{@{}p{0.27\textwidth} p{0.38\textwidth} p{0.29\textwidth}@{}}

    \textbf{Élément} & \textbf{Description} & \textbf{Remarque} \\

    Premier élément & Une cellule qui peut contenir une longue phrase et
      passer à la ligne suivante sans casser la mise en page. & Courte
      remarque \\[2pt]
    Deuxième élément & Une autre description. & Aucune \\[2pt]
    Troisième élément & Encore une description. & Voir
      \autoref{sec:ch1-lists} \\
    \bottomrule
  \end{tabular}
\end{table}

\section{Listes et renvois}\label{sec:ch1-lists}

Une liste à puces :
\begin{itemize}[leftmargin=1.5cm]
  \item premier point ;
  \item second point.
\end{itemize}

Une liste numérotée :
\begin{enumerate}[leftmargin=1.5cm]
  \item première étape ;
  \item seconde étape.
\end{enumerate}

Renvoyez à un élément antérieur grâce à son étiquette, par exemple
\autoref{sec:ch1-elements}.

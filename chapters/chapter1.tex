Reference a table with \autoref{tab:example-table}. Use \texttt{tabularx} so
the table fills the text width and wraps long cells automatically.

  \centering
  \renewcommand{\arraystretch}{1.4}
  \begin{tabularx}{\textwidth}{>{\bfseries}p{3cm} X}
    \toprule
    \rowcolor{tableheadbg}
    \textbf{Column A} & \textbf{Column B} \\
    \midrule
    Row label    & A cell that can hold a long sentence and wrap onto the
                   next line without breaking the table layout. \\
    Another row  & Another description. \\
    \bottomrule
  \end{tabularx}

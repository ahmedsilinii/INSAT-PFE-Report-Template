% ── Chapter Title Format ──────────────────────────────────────────────────────
% Each numbered chapter prints "Chapter N" on its own line, with the
% chapter title in large bold beneath it.
%
% Numbered \chapter uses a generous display style; \chapter* (used by
% \tableofcontents, \listoffigures, etc.) gets its own minimal style so
% the heading does not crowd the list entries beneath it.
\usepackage{titlesec}

\titleformat{\chapter}[display]
  {\normalfont\centering}
  {\large\scshape Chapter\ \thechapter}
  {12pt}
  {\Huge\bfseries}
\titlespacing*{\chapter}{0pt}{60pt}{40pt}

% Unnumbered chapters (ToC, LoF, LoT, Acronyms, Conclusion, Bibliography,
% Appendices) — keep the heading large and clearly separated from its
% content but without the giant grey numeral.
\titleformat{name=\chapter,numberless}[block]
  {\normalfont\raggedright}
  {}
  {0pt}
  {\Huge\bfseries}
\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{32pt}

\usepackage{eso-pic}  % for absolute placement of rotated header on landscape pages

% Helper command: paints a horizontal header along the top of the rotated
% landscape page (page number + chapter name), so the header reads naturally
% when the PDF is viewed in landscape orientation, instead of appearing
% sideways at the edge of the portrait-layout header zone.
\newcommand{\landscapepageheader}{%
  \AddToShipoutPictureFG*{%
    \put(0.5in, 1in){%
      \rotatebox[origin=lb]{90}{%
        \begin{minipage}{\dimexpr\paperheight-2in\relax}%
          \footnotesize\leftmark\hfill\footnotesize\thepage%
          \\[-0.4em]\rule{\linewidth}{0.4pt}%
        \end{minipage}%
      }%
    }%
  }%
}

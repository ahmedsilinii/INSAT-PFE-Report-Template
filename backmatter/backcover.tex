%  Reuses the résumé and abstract texts from frontmatter/ and the titles,

%  nothing to fill in here. If the page overflows, shorten
%  frontmatter/resume.tex and frontmatter/abstract.tex.

\clearpage
\thispagestyle{empty}

% One block per language: title, résumé/abstract text, keywords.
\newcommand{\backcoverblock}[4]{%
  \begin{otherlanguage}{#1}%
    \noindent\rule{\textwidth}{1.2pt}\par
    \noindent{\large\bfseries #2\par}
    \medskip
    \noindent\textbf{\abstractname}\par
    \input{#3}\par
    \smallskip
    \noindent\textbf{\keywordsname} #4\par
  \end{otherlanguage}%
  \bigskip}

\begingroup
\small
\setlength{\parskip}{0.4em}

  \backcoverblock{english}{\TitleEN}{frontmatter/abstract}{\KeywordsEN}
\else
  \backcoverblock{english}{\TitleEN}{frontmatter/abstract}{\KeywordsEN}
  \backcoverblock{french}{\TitleFR}{frontmatter/resume}{\KeywordsFR}
\fi
